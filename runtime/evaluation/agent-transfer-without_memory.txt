function selectEvidence(records, request, revoked) {
  const counts = new Map();
  for (const record of records) {
    if (
      record !== null &&
      typeof record === "object" &&
      typeof record.id === "string" &&
      record.id.length > 0
    ) {
      counts.set(record.id, (counts.get(record.id) || 0) + 1);
    }
  }

  const blocked = new Set(revoked.filter(value => typeof value === "string"));
  const candidates = new Map();

  for (const record of records) {
    if (
      record !== null &&
      typeof record === "object" &&
      !Array.isArray(record) &&
      typeof record.id === "string" &&
      record.id.length > 0 &&
      counts.get(record.id) === 1 &&
      !blocked.has(record.id) &&
      Array.isArray(record.deps) &&
      record.deps.every(dep => typeof dep === "string" && dep.length > 0) &&
      Number.isSafeInteger(record.seq) &&
      record.seq >= 0 &&
      record.state === "verified" &&
      record.core === request.core &&
      record.contract === request.contract &&
      record.evaluator === request.evaluator &&
      record.source === request.source
    ) {
      candidates.set(record.id, record);
    }
  }

  const remaining = new Map();
  const dependents = new Map();
  const ready = [];

  for (const [id, record] of candidates) {
    const dependencies = new Set(record.deps);
    remaining.set(id, dependencies.size);

    if (dependencies.size === 0) ready.push(id);

    for (const dependency of dependencies) {
      if (!candidates.has(dependency)) continue;
      if (!dependents.has(dependency)) dependents.set(dependency, []);
      dependents.get(dependency).push(id);
    }
  }

  const eligible = [];
  for (let index = 0; index < ready.length; index++) {
    const id = ready[index];
    eligible.push(candidates.get(id));

    for (const dependent of dependents.get(id) || []) {
      const count = remaining.get(dependent) - 1;
      remaining.set(dependent, count);
      if (count === 0) ready.push(dependent);
    }
  }

  eligible.sort((a, b) => {
    if (a.seq !== b.seq) return a.seq > b.seq ? -1 : 1;
    return a.id < b.id ? -1 : a.id > b.id ? 1 : 0;
  });

  return eligible.slice(0, request.limit).map(record => record.id);
}
