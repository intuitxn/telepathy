function schedule(events) {
  const items = new Map();
  const dependentsOf = new Map();

  function closure(startId) {
    const out = [];
    const seen = new Set();
    const stack = [startId];
    while (stack.length) {
      const cur = stack.pop();
      if (seen.has(cur)) continue;
      seen.add(cur);
      if (items.has(cur)) out.push(cur);
      const ds = dependentsOf.get(cur);
      if (ds) for (const d of ds) if (!seen.has(d)) stack.push(d);
    }
    return out;
  }

  for (const ev of events) {
    const t = ev && ev.type;
    if (t === 'work') {
      if (items.has(ev.id)) continue;
      const item = {
        id: ev.id,
        deps: ev.deps.slice(),
        priority: ev.priority,
        status: 'pending',
        worker: null,
        revoked: false,
      };
      items.set(ev.id, item);
      for (const d of ev.deps) {
        let s = dependentsOf.get(d);
        if (!s) { s = new Set(); dependentsOf.set(d, s); }
        s.add(ev.id);
      }
    } else if (t === 'claim') {
      const item = items.get(ev.id);
      if (!item || item.status !== 'pending') continue;
      let ok = true;
      for (const d of item.deps) {
        const di = items.get(d);
        if (!di || di.status !== 'done') { ok = false; break; }
      }
      if (ok) {
        for (const o of items.values()) {
          if (o !== item && o.status === 'claimed' && o.worker === ev.worker) { ok = false; break; }
        }
      }
      if (!ok) continue;
      item.status = 'claimed';
      item.worker = ev.worker;
    } else if (t === 'return') {
      const item = items.get(ev.id);
      if (!item || item.status !== 'claimed' || item.worker !== ev.worker) continue;
      let ok = true;
      for (const d of item.deps) {
        const di = items.get(d);
        if (!di || di.status !== 'done') { ok = false; break; }
      }
      if (!ok) continue;
      item.worker = null;
      item.status = ev.ok ? 'done' : 'pending';
    } else if (t === 'correct') {
      if (!items.has(ev.id)) continue;
      for (const id of closure(ev.id)) {
        const item = items.get(id);
        if (!item.revoked) { item.status = 'pending'; item.worker = null; }
      }
    } else if (t === 'revoke') {
      if (!items.has(ev.id)) continue;
      for (const id of closure(ev.id)) {
        const item = items.get(id);
        if (id === ev.id) {
          item.revoked = true;
          item.status = 'revoked';
          item.worker = null;
        } else if (!item.revoked) {
          item.status = 'pending';
          item.worker = null;
        }
      }
    }
  }

  const all = [];
  for (const item of items.values()) {
    all.push({ id: item.id, status: item.status, worker: item.worker });
  }
  all.sort((a, b) => (a.id < b.id ? -1 : a.id > b.id ? 1 : 0));

  const ready = [];
  for (const item of items.values()) {
    if (item.status !== 'pending') continue;
    let ok = true;
    for (const d of item.deps) {
      const di = items.get(d);
      if (!di || di.status !== 'done') { ok = false; break; }
    }
    if (ok) ready.push(item);
  }
  ready.sort((a, b) =>
    a.priority !== b.priority
      ? b.priority - a.priority
      : (a.id < b.id ? -1 : a.id > b.id ? 1 : 0)
  );

  return { states: all, ready: ready.map((i) => i.id) };
}
